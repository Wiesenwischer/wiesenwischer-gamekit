## Anweisungen

### 1. HitWindow erstellen

**Datei:** `Runtime/Data/HitWindow.cs`

```csharp
using System;

namespace Wiesenwischer.GameKit.Combat
{
    /// <summary>
    /// Ein zeitliches Fenster innerhalb einer Attack, definiert durch Start- und End-Frame.
    /// Verwendet für Hit Detection, Cancel Windows, I-Frames etc.
    /// </summary>
    [Serializable]
    public struct HitWindow
    {
        /// <summary>Frame ab dem das Fenster aktiv ist (inklusive).</summary>
        public int startFrame;

        /// <summary>Frame bis zu dem das Fenster aktiv ist (inklusive).</summary>
        public int endFrame;

        /// <summary>Optionaler Identifier für das Fenster (z.B. "primary", "secondary").</summary>
        public string id;

        /// <summary>Dauer des Fensters in Frames.</summary>
        public int Duration => endFrame - startFrame + 1;

        /// <summary>Prüft ob ein gegebener Frame innerhalb dieses Fensters liegt.</summary>
        public bool ContainsFrame(int frame)
        {
            return frame >= startFrame && frame <= endFrame;
        }

        /// <summary>Prüft ob das Fenster gültig ist (startFrame <= endFrame, beide >= 0).</summary>
        public bool IsValid => startFrame >= 0 && endFrame >= startFrame;
    }
}
```

### 2. AttackDefinition erstellen

**Datei:** `Runtime/Data/AttackDefinition.cs`

```csharp
using System;
using UnityEngine;

namespace Wiesenwischer.GameKit.Combat
{
    /// <summary>
    /// Data-driven Definition einer Attack.
    /// Speichert Frame-basiertes Timing, Schaden und Hit Windows.
    /// Serializable (nicht ScriptableObject) für Runtime-Editing und JSON-Persistenz.
    /// </summary>
    [Serializable]
    public class AttackDefinition
    {
        [Header("Identity")]
        public string attackId;

        [Header("Frame Timing")]
        [Tooltip("Gesamtzahl der Frames dieser Attack-Animation")]
        [Min(1)]
        public int totalFrames;

        [Header("Hit Windows")]
        [Tooltip("Primäres Hit Window (wann Schaden verursacht wird)")]
        public HitWindow primaryHitWindow;

        [Header("Combat Data")]
        [Min(0f)]
        public float damage;

        /// <summary>Prüft ob die Definition gültig ist.</summary>
        public bool IsValid =>
            !string.IsNullOrEmpty(attackId) &&
            totalFrames > 0 &&
            primaryHitWindow.IsValid &&
            primaryHitWindow.endFrame < totalFrames;
    }
}
```

**Design-Entscheidungen:**

- **`[Serializable]` statt ScriptableObject:** Runtime Tool muss Daten editieren und als JSON speichern. ScriptableObjects sind im Build nicht editierbar.
- **`totalFrames` explizit gespeichert:** Tool bleibt unabhängig vom AnimationClip. Stabil auch wenn Clip geändert wird.
- **Einzelnes `primaryHitWindow`:** Startet einfach. Erweiterung auf `HitWindow[]` für Multiple Hit Windows in Kap. 13 (Erweiterbarkeit) vorgesehen.

### 3. AttackDatabase erstellen

**Datei:** `Runtime/Data/AttackDatabase.cs`

```csharp
using System.Collections.Generic;
using UnityEngine;

namespace Wiesenwischer.GameKit.Combat
{
    /// <summary>
    /// ScriptableObject-Container für AttackDefinitions.
    /// Dient als Editor-Workflow für die Verwaltung aller Attacks.
    /// Runtime-Zugriff über GetAttack(attackId).
    /// </summary>
    [CreateAssetMenu(
        fileName = "AttackDatabase",
        menuName = "Wiesenwischer/Combat/Attack Database")]
    public class AttackDatabase : ScriptableObject
    {
        [SerializeField]
        private List<AttackDefinition> attacks = new();

        private Dictionary<string, AttackDefinition> _lookup;

        /// <summary>Alle registrierten Attacks.</summary>
        public IReadOnlyList<AttackDefinition> Attacks => attacks;

        /// <summary>
        /// Sucht eine AttackDefinition anhand ihrer ID.
        /// Baut beim ersten Aufruf einen Lookup-Cache auf.
        /// </summary>
        public AttackDefinition GetAttack(string attackId)
        {
            if (_lookup == null)
                BuildLookup();

            _lookup.TryGetValue(attackId, out var attack);
            return attack;
        }

        /// <summary>Gibt true zurück wenn eine Attack mit dieser ID existiert.</summary>
        public bool HasAttack(string attackId)
        {
            if (_lookup == null)
                BuildLookup();

            return _lookup.ContainsKey(attackId);
        }

        /// <summary>Rebuild des Lookup-Cache (z.B. nach Runtime-Änderungen).</summary>
        public void RebuildLookup() => BuildLookup();

        private void BuildLookup()
        {
            _lookup = new Dictionary<string, AttackDefinition>();
            foreach (var attack in attacks)
            {
                if (attack != null && !string.IsNullOrEmpty(attack.attackId))
                    _lookup[attack.attackId] = attack;
            }
        }

        private void OnEnable()
        {
            _lookup = null; // Lazy rebuild
        }
    }
}
```
