## Ziel

Separate Mapping Layer zwischen Animator-States und AttackDefinitions. Animation kann geändert werden ohne Gameplay-Daten zu zerstören. Mehrere Animationen können die gleiche AttackDefinition nutzen.

---

## Anweisungen

### 1. AttackMapping erstellen

**Datei:** `Runtime/Data/AttackMapping.cs`

```csharp
using System;

namespace Wiesenwischer.GameKit.Combat
{
    /// <summary>
    /// Mapping zwischen einem Animator State und einer AttackDefinition.
    /// Entkoppelt Animation (Visual) von Gameplay Timing (Data).
    ///
    /// Warum Mapping Layer:
    /// - Animation kann geändert werden ohne Gameplay Daten zu zerstören
    /// - Mehrere Animationen können gleiche AttackDefinition nutzen
    /// - Cleaner Separation of Concerns
    /// </summary>
    [Serializable]
    public class AttackMapping
    {
        /// <summary>Name des Animator-States (z.B. "Slash_01", "Thrust_Heavy").</summary>
        public string animationStateName;

        /// <summary>ID der zugehörigen AttackDefinition.</summary>
        public string attackId;

        /// <summary>Prüft ob das Mapping gültig ist.</summary>
        public bool IsValid =>
            !string.IsNullOrEmpty(animationStateName) &&
            !string.IsNullOrEmpty(attackId);
    }
}
```

### 2. AttackMappingDatabase erstellen

**Datei:** `Runtime/Data/AttackMappingDatabase.cs`

```csharp
using System.Collections.Generic;
using UnityEngine;

namespace Wiesenwischer.GameKit.Combat
{
    /// <summary>
    /// ScriptableObject-Container für AttackMappings.
    /// Stellt den Lookup animationStateName → attackId bereit.
    /// </summary>
    [CreateAssetMenu(
        fileName = "AttackMappingDatabase",
        menuName = "Wiesenwischer/Combat/Attack Mapping Database")]
    public class AttackMappingDatabase : ScriptableObject
    {
        [SerializeField]
        private List<AttackMapping> mappings = new();

        private Dictionary<string, string> _animToAttackLookup;
        private Dictionary<string, List<string>> _attackToAnimLookup;

        /// <summary>Alle registrierten Mappings.</summary>
        public IReadOnlyList<AttackMapping> Mappings => mappings;

        /// <summary>
        /// Gibt die AttackDefinition-ID für einen Animator-State zurück.
        /// Null wenn kein Mapping existiert.
        /// </summary>
        public string GetAttackIdForAnimation(string animationStateName)
        {
            if (_animToAttackLookup == null)
                BuildLookup();

            _animToAttackLookup.TryGetValue(animationStateName, out var attackId);
            return attackId;
        }

        /// <summary>
        /// Gibt alle Animator-States zurück, die einer AttackDefinition zugeordnet sind.
        /// </summary>
        public IReadOnlyList<string> GetAnimationsForAttack(string attackId)
        {
            if (_attackToAnimLookup == null)
                BuildLookup();

            return _attackToAnimLookup.TryGetValue(attackId, out var anims)
                ? anims
                : System.Array.Empty<string>();
        }

        /// <summary>Rebuild des Lookup-Cache.</summary>
        public void RebuildLookup() => BuildLookup();

        private void BuildLookup()
        {
            _animToAttackLookup = new Dictionary<string, string>();
            _attackToAnimLookup = new Dictionary<string, List<string>>();

            foreach (var mapping in mappings)
            {
                if (mapping == null || !mapping.IsValid)
                    continue;

                _animToAttackLookup[mapping.animationStateName] = mapping.attackId;

                if (!_attackToAnimLookup.TryGetValue(mapping.attackId, out var list))
                {
                    list = new List<string>();
                    _attackToAnimLookup[mapping.attackId] = list;
                }
                list.Add(mapping.animationStateName);
            }
        }

        private void OnEnable()
        {
            _animToAttackLookup = null;
            _attackToAnimLookup = null;
        }
    }
}
```
