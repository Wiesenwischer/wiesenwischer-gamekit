## Anweisungen

### 1. AttackPersistence erstellen

**Datei:** `Runtime/Core/AttackPersistence.cs`

```csharp
using System.Collections.Generic;
using System.IO;
using UnityEngine;

namespace Wiesenwischer.GameKit.Combat
{
    /// <summary>
    /// JSON-basierte Persistenz für AttackDefinitions.
    /// Speichert und lädt Attack-Daten im StreamingAssets-Verzeichnis.
    ///
    /// Designed für Runtime-Nutzung im Tuning Tool:
    /// - JsonUtility für Serialisierung (Unity-kompatibel)
    /// - StreamingAssets als Speicherort (im Build zugänglich)
    /// </summary>
    public static class AttackPersistence
    {
        private const string DefaultSubFolder = "CombatData";
        private const string FileExtension = ".json";

        /// <summary>
        /// Speichert eine einzelne AttackDefinition als JSON-Datei.
        /// </summary>
        /// <param name="attack">Die zu speichernde AttackDefinition.</param>
        /// <param name="subFolder">Unterordner in StreamingAssets (default: "CombatData").</param>
        public static void SaveAttack(AttackDefinition attack, string subFolder = DefaultSubFolder)
        {
            string directory = GetDirectory(subFolder);
            EnsureDirectoryExists(directory);

            string filePath = Path.Combine(directory, attack.attackId + FileExtension);
            string json = JsonUtility.ToJson(attack, prettyPrint: true);
            File.WriteAllText(filePath, json);
        }

        /// <summary>
        /// Lädt eine einzelne AttackDefinition aus einer JSON-Datei.
        /// </summary>
        /// <param name="attackId">ID der Attack (= Dateiname ohne Erweiterung).</param>
        /// <param name="subFolder">Unterordner in StreamingAssets.</param>
        /// <returns>Die geladene AttackDefinition oder null wenn nicht gefunden.</returns>
        public static AttackDefinition LoadAttack(string attackId, string subFolder = DefaultSubFolder)
        {
            string filePath = Path.Combine(GetDirectory(subFolder), attackId + FileExtension);

            if (!File.Exists(filePath))
                return null;

            string json = File.ReadAllText(filePath);
            var attack = new AttackDefinition();
            JsonUtility.FromJsonOverwrite(json, attack);
            return attack;
        }

        /// <summary>
        /// Speichert alle Attacks einer AttackDatabase als einzelne JSON-Dateien.
        /// </summary>
        public static void SaveAll(AttackDatabase database, string subFolder = DefaultSubFolder)
        {
            foreach (var attack in database.Attacks)
            {
                if (attack != null && !string.IsNullOrEmpty(attack.attackId))
                    SaveAttack(attack, subFolder);
            }
        }

        /// <summary>
        /// Lädt alle JSON-Dateien aus dem angegebenen Ordner.
        /// </summary>
        /// <param name="subFolder">Unterordner in StreamingAssets.</param>
        /// <returns>Liste aller geladenen AttackDefinitions.</returns>
        public static List<AttackDefinition> LoadAll(string subFolder = DefaultSubFolder)
        {
            var result = new List<AttackDefinition>();
            string directory = GetDirectory(subFolder);

            if (!Directory.Exists(directory))
                return result;

            foreach (var file in Directory.GetFiles(directory, "*" + FileExtension))
            {
                string json = File.ReadAllText(file);
                var attack = new AttackDefinition();
                JsonUtility.FromJsonOverwrite(json, attack);

                if (attack.IsValid)
                    result.Add(attack);
            }

            return result;
        }

        /// <summary>
        /// Löscht die JSON-Datei einer Attack.
        /// </summary>
        public static bool DeleteAttack(string attackId, string subFolder = DefaultSubFolder)
        {
            string filePath = Path.Combine(GetDirectory(subFolder), attackId + FileExtension);
            if (!File.Exists(filePath))
                return false;

            File.Delete(filePath);
            return true;
        }

        /// <summary>
        /// Prüft ob eine JSON-Datei für die gegebene Attack existiert.
        /// </summary>
        public static bool Exists(string attackId, string subFolder = DefaultSubFolder)
        {
            string filePath = Path.Combine(GetDirectory(subFolder), attackId + FileExtension);
            return File.Exists(filePath);
        }

        private static string GetDirectory(string subFolder)
        {
            return Path.Combine(Application.streamingAssetsPath, subFolder);
        }

        private static void EnsureDirectoryExists(string path)
        {
            if (!Directory.Exists(path))
                Directory.CreateDirectory(path);
        }
    }
}
```

**Design-Entscheidungen:**

- **StreamingAssets:** Im Build les- und schreibbar (anders als Resources). Perfekt für Runtime-Tool-Daten.
- **Einzelne Dateien pro Attack:** Einfaches Dateisystem, weniger Merge-Konflikte, einzeln exportierbar.
- **JsonUtility:** Unity-native, kein externer JSON-Parser nötig. Unterstützt `[Serializable]` Klassen.
- **Statische Klasse:** Kein State, reine Utility. Kann von überall aufgerufen werden.

## Verifikations-Checkliste

- [ ] `SaveAttack()` erstellt JSON-Datei in StreamingAssets/CombatData/
- [ ] `LoadAttack()` liest JSON und gibt AttackDefinition zurück
- [ ] `SaveAll()` speichert alle Attacks einer Database
- [ ] `LoadAll()` lädt alle JSON-Dateien aus dem Ordner
- [ ] `DeleteAttack()` entfernt eine JSON-Datei
- [ ] `Exists()` prüft Datei-Existenz
- [ ] Directory wird automatisch erstellt wenn nicht vorhanden
- [ ] Ungültige Attacks (IsValid == false) werden beim Laden gefiltert
- [ ] Keine Compiler-Fehler
